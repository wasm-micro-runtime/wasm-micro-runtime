cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico2_w CACHE STRING "Board")
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(stackless_http_pico C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()

if(NOT WAMR_ROOT OR NOT HANDLER_INCLUDE)
    message(FATAL_ERROR "Set WAMR_ROOT and HANDLER_INCLUDE (generated handler.h directory)")
endif()
include(../embedded-wamr.cmake)

if(P3_WASI_HTTP)
    set(http_sources ../wasi_http.c ../wasi_http_server.c)
    set(stack_size 8192)
else()
    set(http_sources ../http.c)
    set(stack_size 4096)
endif()
add_executable(stackless_http ${WAMR_RUNTIME_LIB_SOURCE} ../p3.c ../vm.c ../task.c ${http_sources} pico_server.c)
if(P3_WASI_HTTP)
    target_compile_definitions(stackless_http PRIVATE P3_WASI_HTTP)
    # The default RP2350 core-zero stack lives in a 4 KiB scratch bank.
    # Put the larger, budgeted stack in main SRAM for nested canonical calls.
    set(sdk_linker "${PICO_SDK_PATH}/src/rp2_common/pico_crt0/rp2350/memmap_default.ld")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${sdk_linker}")
    file(READ "${sdk_linker}" linker)
    set(sdk_stack "KEEP(*(.stack*))\n    } > SCRATCH_Y")
    set(sdk_stack_top "__StackTop = ORIGIN(SCRATCH_Y) + LENGTH(SCRATCH_Y);")
    string(FIND "${linker}" "${sdk_stack}" stack_section)
    string(FIND "${linker}" "${sdk_stack_top}" stack_top)
    if(stack_section EQUAL -1 OR stack_top EQUAL -1)
        message(FATAL_ERROR "Unsupported RP2350 SDK linker layout")
    endif()
    string(REPLACE "${sdk_stack}" "KEEP(*(.stack*))\n    } > RAM" linker "${linker}")
    string(REPLACE "${sdk_stack_top}" "__StackTop = ADDR(.stack_dummy) + SIZEOF(.stack_dummy);" linker "${linker}")
    set(linker_path "${CMAKE_CURRENT_BINARY_DIR}/stackless_http.ld")
    file(GENERATE OUTPUT "${linker_path}" CONTENT "${linker}")
    pico_set_linker_script(stackless_http "${linker_path}")
endif()
target_include_directories(stackless_http PRIVATE . .. ${HANDLER_INCLUDE}
    ${WAMR_ROOT}/core/iwasm/include)
target_compile_definitions(stackless_http PRIVATE P3_WAMR PICO_STACK_SIZE=${stack_size} PICO_HEAP_SIZE=16384)
target_link_libraries(stackless_http PRIVATE pico_stdlib pico_cyw43_arch_lwip_threadsafe_background)
set_property(TARGET stackless_http PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
# GCC's LTO pass cannot see the SDK's ld --wrap references. Keep the wrappers
# referenced by the firmware visible to the link plugin.
foreach(symbol printf vprintf puts putchar getchar snprintf vsnprintf ceil floor trunc)
    target_link_options(stackless_http PRIVATE "LINKER:-u,__wrap_${symbol}")
endforeach()
pico_enable_stdio_usb(stackless_http 1)
pico_enable_stdio_uart(stackless_http 1)
pico_add_extra_outputs(stackless_http)
