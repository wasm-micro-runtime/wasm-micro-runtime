cmake_minimum_required(VERSION 3.20)
project(stackless_http C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

if(NOT P3_RUNTIME_LIBRARY OR NOT P3_RUNTIME_INCLUDE OR NOT P3_HANDLER_INCLUDE)
    message(FATAL_ERROR "Set P3_RUNTIME_LIBRARY, P3_RUNTIME_INCLUDE and P3_HANDLER_INCLUDE")
endif()

add_executable(stackless_http p3.c vm.c task.c http.c server.c)
add_executable(component_check p3.c vm.c check.c)
target_compile_definitions(component_check PRIVATE P3_HANDLES=128)
foreach(target stackless_http component_check)
    target_include_directories(${target} PRIVATE ${P3_RUNTIME_INCLUDE} ${P3_HANDLER_INCLUDE})
    target_link_libraries(${target} PRIVATE ${P3_RUNTIME_LIBRARY})
    if(P3_WAMR)
        target_compile_definitions(${target} PRIVATE P3_WAMR)
    endif()
    if(MSVC)
        target_compile_definitions(${target} PRIVATE LIBWASM_STATIC WASM_API_EXTERN=)
        target_compile_options(${target} PRIVATE /W3)
        target_link_libraries(${target} PRIVATE ws2_32 psapi userenv bcrypt ntdll advapi32)
        target_link_options(${target} PRIVATE /INCREMENTAL:NO /OPT:REF /OPT:ICF)
        if(P3_ASAN)
            target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
        if(P3_ASAN)
            target_compile_options(${target} PRIVATE -fsanitize=address -fno-omit-frame-pointer -g)
            target_link_options(${target} PRIVATE -fsanitize=address)
        endif()
        if(APPLE)
            if(NOT P3_WAMR)
                target_link_libraries(${target} PRIVATE "-framework Security" "-framework CoreFoundation" iconv)
            endif()
            target_link_options(${target} PRIVATE -Wl,-dead_strip)
        else()
            target_link_libraries(${target} PRIVATE m pthread dl)
            target_link_options(${target} PRIVATE -Wl,--gc-sections)
        endif()
    endif()
    if(NOT P3_ASAN)
        set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
    endif()
endforeach()
