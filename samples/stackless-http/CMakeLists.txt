cmake_minimum_required(VERSION 3.20)
project(stackless_http C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

if(NOT P3_RUNTIME_LIBRARY OR NOT P3_RUNTIME_INCLUDE OR NOT P3_HANDLER_INCLUDE)
    message(FATAL_ERROR "Set P3_RUNTIME_LIBRARY, P3_RUNTIME_INCLUDE and P3_HANDLER_INCLUDE")
endif()

add_executable(stackless_http p3.c vm.c task.c http.c server.c)
add_executable(component_check p3.c vm.c check.c)
add_executable(wasi_http_check p3.c vm.c task.c wasi_http.c wasi_http_check.c)
target_compile_definitions(wasi_http_check PRIVATE P3_WASI_HTTP P3_WASI_HTTP_HOSTED P3_IMPORTS=128 HTTP_FIELD_BYTES=4096)
target_compile_definitions(component_check PRIVATE P3_HANDLES=128)
set(targets stackless_http component_check wasi_http_check)
if(P3_WASI_HANDLER_INCLUDE)
    add_executable(wasi_http_server p3.c vm.c task.c wasi_http.c wasi_http_io.c wasi_http_server.c server.c)
    target_include_directories(wasi_http_server PRIVATE ${P3_WASI_HANDLER_INCLUDE})
    target_compile_definitions(wasi_http_server PRIVATE P3_WASI_HTTP P3_WASI_HTTP_HOSTED P3_IMPORTS=128)
    list(APPEND targets wasi_http_server)
endif()
if(P3_H2_HANDLER_INCLUDE)
    if(NOT P3_NGHTTP2_INCLUDE OR NOT P3_NGHTTP2_LIBRARY)
        message(FATAL_ERROR "Set P3_NGHTTP2_INCLUDE and P3_NGHTTP2_LIBRARY")
    endif()
    add_executable(wasi_h2_server p3.c vm.c task.c wasi_http.c wasi_http_io.c h2.c h2_server.c)
    target_include_directories(wasi_h2_server PRIVATE ${P3_H2_HANDLER_INCLUDE} ${P3_NGHTTP2_INCLUDE})
    target_link_libraries(wasi_h2_server PRIVATE ${P3_NGHTTP2_LIBRARY})
    target_compile_definitions(wasi_h2_server PRIVATE P3_WASI_HTTP NGHTTP2_STATICLIB NGHTTP2_NO_SSIZE_T
        P3_HANDLES=240 HTTP_RESOURCE_SLOTS=64 HTTP_PENDING_DROPS=48
        H2_STREAMS=16 H2_PER_CONNECTION=8 H2_CONNECTIONS=4 H2_CONNECTION_HEAP=65536)
    list(APPEND targets wasi_h2_server)
endif()
foreach(target IN LISTS targets)
    target_include_directories(${target} PRIVATE ${P3_RUNTIME_INCLUDE} ${P3_HANDLER_INCLUDE})
    target_link_libraries(${target} PRIVATE ${P3_RUNTIME_LIBRARY})
    if(P3_WAMR)
        target_compile_definitions(${target} PRIVATE P3_WAMR)
    endif()
    if(MSVC)
        target_compile_definitions(${target} PRIVATE LIBWASM_STATIC WASM_API_EXTERN=)
        target_compile_options(${target} PRIVATE /W3)
        target_link_libraries(${target} PRIVATE ws2_32 psapi userenv bcrypt ntdll advapi32)
        target_link_options(${target} PRIVATE /INCREMENTAL:NO /OPT:REF /OPT:ICF)
        if(P3_ASAN)
            target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
        if(P3_ASAN)
            target_compile_options(${target} PRIVATE -fsanitize=address -fno-omit-frame-pointer -g)
            target_link_options(${target} PRIVATE -fsanitize=address)
        endif()
        if(APPLE)
            if(NOT P3_WAMR)
                target_link_libraries(${target} PRIVATE "-framework Security" "-framework CoreFoundation" iconv)
            endif()
            target_link_options(${target} PRIVATE -Wl,-dead_strip)
        else()
            target_link_libraries(${target} PRIVATE m pthread dl)
            target_link_options(${target} PRIVATE -Wl,--gc-sections)
        endif()
    endif()
    if(NOT P3_ASAN)
        set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
    endif()
endforeach()
