cmake_minimum_required(VERSION 3.20)
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_SYSTEM_PROCESSOR arm)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
set(CMAKE_C_COMPILER arm-none-eabi-gcc)
set(CMAKE_ASM_COMPILER arm-none-eabi-gcc)
project(stackless_http_m33 C ASM)
set(CMAKE_C_STANDARD 11)
if(NOT WAMR_ROOT OR NOT HANDLER_INCLUDE)
    message(FATAL_ERROR "Set WAMR_ROOT and HANDLER_INCLUDE")
endif()
include(../embedded-wamr.cmake)
set(sim_source sim.c)
set(heap_size 16384)
if(P3_WASI_H2)
    if(NOT P3_NGHTTP2_INCLUDE OR NOT P3_NGHTTP2_CONFIG_INCLUDE OR NOT P3_NGHTTP2_LIBRARY)
        message(FATAL_ERROR "Set P3_NGHTTP2_INCLUDE, P3_NGHTTP2_CONFIG_INCLUDE and P3_NGHTTP2_LIBRARY")
    endif()
    set(http_sources ../wasi_http.c ../wasi_http_io.c ../h2.c)
    set(sim_source h2_sim.c)
    set(heap_size 65536)
elseif(P3_WASI_HTTP)
    set(http_sources ../wasi_http.c ../wasi_http_io.c ../wasi_http_server.c)
else()
    set(http_sources ../http.c)
endif()
add_executable(stackless_m33 ${WAMR_RUNTIME_LIB_SOURCE} ../p3.c ../vm.c ../task.c ${http_sources}
    startup.c ${sim_source})
if(P3_WASI_HTTP OR P3_WASI_H2)
    target_compile_definitions(stackless_m33 PRIVATE P3_WASI_HTTP)
    target_link_options(stackless_m33 PRIVATE "LINKER:--defsym,__p3_stack_size=8192")
endif()
if(P3_WASI_H2)
    target_include_directories(stackless_m33 PRIVATE ${P3_NGHTTP2_INCLUDE} ${P3_NGHTTP2_CONFIG_INCLUDE})
    target_compile_definitions(stackless_m33 PRIVATE NGHTTP2_STATICLIB NGHTTP2_NO_SSIZE_T)
    target_link_libraries(stackless_m33 PRIVATE ${P3_NGHTTP2_LIBRARY})
    target_link_options(stackless_m33 PRIVATE "LINKER:--defsym,__p3_heap_size=${heap_size}")
endif()
target_include_directories(stackless_m33 PRIVATE .. ${HANDLER_INCLUDE} ${WAMR_ROOT}/core/iwasm/include)
target_compile_definitions(stackless_m33 PRIVATE P3_WAMR P3_QEMU PICO_HEAP_SIZE=${heap_size})
target_compile_options(stackless_m33 PRIVATE -mcpu=cortex-m33 -mthumb -mfloat-abi=soft -ffunction-sections -fdata-sections)
target_link_options(stackless_m33 PRIVATE -mcpu=cortex-m33 -mthumb -mfloat-abi=soft -nostartfiles --specs=nosys.specs
    -T${CMAKE_CURRENT_LIST_DIR}/m33.ld -Wl,--gc-sections,-Map=stackless_m33.map)
target_link_libraries(stackless_m33 PRIVATE m)
set_target_properties(stackless_m33 PROPERTIES SUFFIX .elf INTERPROCEDURAL_OPTIMIZATION TRUE)
set_property(TARGET stackless_m33 APPEND PROPERTY LINK_DEPENDS ${CMAKE_CURRENT_LIST_DIR}/m33.ld)
